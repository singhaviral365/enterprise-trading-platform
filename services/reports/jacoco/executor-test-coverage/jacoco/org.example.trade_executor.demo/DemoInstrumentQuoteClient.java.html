package org.example.trade_executor.demo;

import org.example.trade_executor.client.FauxnanceQuoteClient;
import org.example.trade_executor.client.LiveQuote;
import org.example.trade_executor.client.PriceHistory;
import org.example.trade_executor.client.QuoteClient;
import org.example.trade_executor.client.QuoteDetails;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.context.annotation.Primary;
import org.springframework.stereotype.Component;

import java.math.BigDecimal;
import java.time.Clock;
import java.time.Instant;
import java.time.LocalDate;
import java.util.ArrayList;
import java.util.Collection;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Locale;
import java.util.Map;
import java.util.Optional;

/**
 * DEMO ONLY. Adds one instrument that Fauxnance does not know, whose price swings between two values (295 and 305 US dollars by default)
 * on a fixed beat, so a price alert and an automatic strategy can be shown firing again and again.
 *
 * <p>It wraps the real quote client: every other symbol is answered by Fauxnance exactly as before, and the demo symbol is taken out of
 * the request so Fauxnance is never asked about it. The market poller, the market board, the candles route and order pricing all go
 * through {@link QuoteClient}, so the demo instrument appears in each of them with no other change.
 *
 * <p>The price depends on the clock, not on who asks: the value flips every {@code period-seconds} (set it to the poll interval, 60 by
 * default), low in even periods and high in odd ones.
 *
 * <p><b>To remove it:</b> set {@code executor.demo-instrument.enabled: false} in config/application.yml (this class and
 * {@link DemoInstrumentSeeder} are then not created), and run databases/PostgreSQL/demo/remove-demo-instrument.sql to delete the row. Or delete
 * this package, the three {@code demo.instrument.*} lines in application.properties and the yml block: nothing else refers to it.
 */
@Component
@Primary
@ConditionalOnProperty(name = "demo.instrument.enabled", havingValue = "true")
public class DemoInstrumentQuoteClient implements QuoteClient {

    static final String CURRENCY = "USD";

    private final QuoteClient real;
    private final String symbol;
    private final BigDecimal low;
    private final BigDecimal high;
    private final long periodSeconds;
    private final Clock clock;

    @Autowired
    public DemoInstrumentQuoteClient(
            FauxnanceQuoteClient real,
            @Value("${demo.instrument.symbol}") String symbol,
            @Value("${demo.instrument.low}") BigDecimal low,
            @Value("${demo.instrument.high}") BigDecimal high,
            @Value("${demo.instrument.period-seconds}") long periodSeconds) {
        this(real, symbol, low, high, periodSeconds, Clock.systemUTC());
    }

    DemoInstrumentQuoteClient(QuoteClient real, String symbol, BigDecimal low, BigDecimal high, long periodSeconds, Clock clock) {
        this.real = real;
        this.symbol = symbol.trim().toUpperCase(Locale.ROOT);
        this.low = low;
        this.high = high;
        this.periodSeconds = Math.max(1L, periodSeconds);
        this.clock = clock;
    }

    public String symbol() {
        return symbol;
    }

    /** The price now: low in even periods, high in odd ones. */
    BigDecimal priceAt(Instant moment) {
        long period = Math.floorDiv(moment.getEpochSecond(), periodSeconds);
        return Math.floorMod(period, 2L) == 0 ? low : high;
    }

    private boolean isDemo(String candidate) {
        return candidate != null && symbol.equals(candidate.trim().toUpperCase(Locale.ROOT));
    }

    private LiveQuote demoQuote() {
        return new LiveQuote(symbol, priceAt(clock.instant()), CURRENCY);
    }

    @Override
    public Optional<LiveQuote> getQuote(String requested) {
        return isDemo(requested) ? Optional.of(demoQuote()) : real.getQuote(requested);
    }

    @Override
    public Map<String, LiveQuote> getQuotes(Collection<String> symbols) {
        Map<String, LiveQuote> quotes = new LinkedHashMap<>();
        boolean wanted = false;
        List<String> others = new ArrayList<>();
        if (symbols != null) {
            for (String candidate : symbols) {
                if (isDemo(candidate)) {
                    wanted = true;
                } else {
                    others.add(candidate);
                }
            }
        }
        if (!others.isEmpty()) {
            quotes.putAll(real.getQuotes(others));
        }
        if (wanted) {
            quotes.put(symbol, demoQuote());
        }
        return quotes;
    }

    @Override
    public Map<String, QuoteDetails> getQuoteDetails(Collection<String> symbols) {
        Map<String, QuoteDetails> details = new LinkedHashMap<>();
        boolean wanted = false;
        List<String> others = new ArrayList<>();
        if (symbols != null) {
            for (String candidate : symbols) {
                if (isDemo(candidate)) {
                    wanted = true;
                } else {
                    others.add(candidate);
                }
            }
        }
        if (!others.isEmpty()) {
            details.putAll(real.getQuoteDetails(others));
        }
        if (wanted) {
            Instant now = clock.instant();
            BigDecimal price = priceAt(now);
            BigDecimal previous = price.equals(low) ? high : low;
            Map<String, Object> fields = new LinkedHashMap<>();
            fields.put("symbol", symbol);
            fields.put("price", price);
            fields.put("currency", CURRENCY);
            fields.put("previousClose", previous);
            fields.put("change", price.subtract(previous));
            fields.put("changePercent", price.subtract(previous).multiply(BigDecimal.valueOf(100)).divide(previous, 4, java.math.RoundingMode.HALF_UP));
            fields.put("bid", price);
            fields.put("ask", price);
            fields.put("marketState", "REGULAR");
            fields.put("source", "demo");
            fields.put("stale", false);
            fields.put("asOf", now.toString());
            details.put(symbol, new QuoteDetails(symbol, price, fields));
        }
        return details;
    }

    @Override
    public PriceHistory getCandles(String requested, LocalDate from, LocalDate to, String interval) {
        if (!isDemo(requested)) {
            return real.getCandles(requested, from, to, interval);
        }
        // A year of daily candles that swing between the two prices, so the chart has something to draw.
        List<PriceHistory.Candle> candles = new ArrayList<>();
        BigDecimal previous = low;
        for (LocalDate day = from; !day.isAfter(to); day = day.plusDays(1)) {
            BigDecimal close = (day.toEpochDay() % 2 == 0) ? low : high;
            candles.add(new PriceHistory.Candle(day.toString(), previous, previous.max(close), previous.min(close), close, 1_000L, false));
            previous = close;
        }
        return new PriceHistory(symbol, CURRENCY, interval, candles);
    }
}
