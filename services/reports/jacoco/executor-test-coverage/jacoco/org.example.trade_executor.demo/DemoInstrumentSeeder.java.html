package org.example.trade_executor.demo;

import org.slf4j.Logger;
import org.slf4j.LoggerFactory;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.boot.ApplicationArguments;
import org.springframework.boot.ApplicationRunner;
import org.springframework.boot.autoconfigure.condition.ConditionalOnProperty;
import org.springframework.jdbc.core.JdbcTemplate;
import org.springframework.stereotype.Component;

import java.util.Locale;

/**
 * DEMO ONLY. Makes sure the demo instrument has a row in the instrument table when the executor starts, so it shows on the market board and
 * can be put on a watchlist, given a price alert, bought, or used in a strategy. It never changes an existing row.
 * See {@link DemoInstrumentQuoteClient} for how to remove the demo.
 */
@Component
@ConditionalOnProperty(name = "demo.instrument.enabled", havingValue = "true")
public class DemoInstrumentSeeder implements ApplicationRunner {

    private static final Logger LOGGER = LoggerFactory.getLogger(DemoInstrumentSeeder.class);

    private final JdbcTemplate jdbc;
    private final String symbol;
    private final String name;

    public DemoInstrumentSeeder(
            JdbcTemplate jdbc,
            @Value("${demo.instrument.symbol}") String symbol,
            @Value("${demo.instrument.name}") String name) {
        this.jdbc = jdbc;
        this.symbol = symbol.trim().toUpperCase(Locale.ROOT);
        this.name = name;
    }

    @Override
    public void run(ApplicationArguments args) {
        int added = jdbc.update(
                "INSERT INTO instrument (instrument_ticker, instrument_name, asset_class, instrument_status, currency) "
                        + "SELECT ?, ?, 'EQUITY', 'TRADING', 'USD' "
                        + "WHERE NOT EXISTS (SELECT 1 FROM instrument WHERE instrument_ticker = ?)",
                symbol, name, symbol);
        if (added > 0) {
            LOGGER.info("Demo instrument {} added to the instrument table (its price is made up; Fauxnance does not know it)", symbol);
        } else {
            LOGGER.info("Demo instrument {} is on (its price is made up; Fauxnance does not know it)", symbol);
        }
    }
}
